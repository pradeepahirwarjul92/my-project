01.  Reverse a string without using built-in reverse.
Ans- 

Solution 1: Using Character Array (Most Expected)

public class ReverseString {
    public static void main(String[] args) {
        String str = "Pradeep";

        char[] arr = str.toCharArray();

        int left = 0;
        int right = arr.length - 1;

        while (left < right) {
            char temp = arr[left];
            arr[left] = arr[right];
            arr[right] = temp;

            left++;
            right--;
        }

        System.out.println(new String(arr));
    }
}
	
	Complexity
Time: O(n)
Space: O(n) (char array)


Solution 2: Using Loop (Frequently Asked)

public class ReverseString {
    public static void main(String[] args) {
        String str = "Pradeep";
        String reversed = "";

        for (int i = str.length() - 1; i >= 0; i--) {
            reversed += str.charAt(i);
        }

        System.out.println(reversed);
    }
}

Follow-up

Interviewer may ask:

What's wrong with this approach?
Ans - String is immutable.
Every concatenation creates a new object.
Time complexity becomes O(n²).

}

Complexity

Solution 4: Recursive Solution (Senior-Level Follow-up)
public class ReverseString {

    public static String reverse(String str) {

        if (str.isEmpty()) {
            return str;
        }

        return reverse(str.substring(1)) + str.charAt(0);
    }

    public static void main(String[] args) {
        System.out.println(reverse("Pradeep"));
    }
}


Complexity
Time: O(n²)
Space: O(n)


2. Check whether a string is palindrome.
Ans - 
Solution 1: Two Pointer Approach (Most Expected)

This is usually the preferred interview solution.

public class PalindromeCheck {

    public static boolean isPalindrome(String str) {

        int left = 0;
        int right = str.length() - 1;

        while (left < right) {

            if (str.charAt(left) != str.charAt(right)) {
                return false;
            }

            left++;
            right--;
        }

        return true;
    }

    public static void main(String[] args) {

        String str = "MADAM";

        if (isPalindrome(str)) {
            System.out.println("Palindrome");
        } else {
            System.out.println("Not Palindrome");
        }
    }
}
Complexity
Time  : O(n)
Space : O(1)

Use two pointers:
- One starts from the beginning.
- One starts from the end.
- Compare characters.
- If any mismatch occurs, return false.
- If all characters match, return true.

Time Complexity: O(n)
Space Complexity: O(1)

                .mapToObj(c -> (char) c)

19. Find Longest Palindromic Substring

package com.heg.super500;
public class Q19_LongestPalindromiceSubstring {
    public static String longestPalindrome(String s){
        if(s == null || s.length()<2){
            return s;
        }        
        int start=0;
        int end=0;        
        for(int i =0 ; i< s.length(); i++){
            int len1=expand(s,i, i); //odd length           
            int len2=expand(s, i,i+1); //even length  
            int len= Math.max(len1, len2);           
             System.out.println("len1 - "+ len1 +  ", len2 -" +len2 + " =  Len  ::  " +len + ", Start - " +start + " end " +end);            
            if(len > end - start){
                start= i - (len -1)/2;
                end = i +len/2;
            }  
            System.out.println("len1 - "+ len1 +  ", len2 -" +len2 + " =  Len  ::  " +len + ", Start - " +start + " end " +end);
            System.out.println("");            
        }        
        return s.substring(start, end+1);  
    }

    private static int expand(String s, int left, int right) {
        while(left >= 0 && 
                right < s.length() &&
                s.charAt(left)==s.charAt(right)){
            right++;
            left --;            
        }
        return right - left -1 ;
    }
    
    public static void main(String[] args) {
        String str= "babad";
        System.out.println(longestPalindrome(str));
    }    
}

21. Check Balanced Parentheses
package com.heg.super500;
import java.util.Stack;

public class Q21_BalancedParentheses {
    public static boolean isBalanced(String str) {
        Stack<Character> stack = new Stack();
        for (char ch : str.toCharArray()) {
            if (ch == '(' || ch == '{' || ch == '[') {
                stack.push(ch);
            } else {
                if (stack.isEmpty()) {
                    return false;
                }
                char top = stack.pop();
                if ((ch == ')' && top != '(')
                        || (ch == '}' && top != '{')
                        || (ch == ']' && top != '[')) {
                    return false;
                }
            }
        }
        return stack.isEmpty();
    }

    public static void main(String[] args) {
        System.out.println(isBalanced("{[()]}"));

    }
}

26. Sort Characters in String
package com.heg.super500;
public class Q26_SortUsingBubbleSort {
    public static String bubbleSort(String str){
        if(str.length()<1){
            return str;
        }        
        char[] arr= str.toCharArray();        
        for(int i=0; i< arr.length-1; i++){
            for(int j=0 ; j< arr.length-i-1 ;j++){
                if(arr[j]> arr[j+1]){
                    char temp = arr[j];
                    arr[j]=arr[j+1];
                    arr[j+1]=temp;
                }
            }
        }
        return new String(arr);    
    }    
    public static void main(String[] args) {
        String str="dcba";
        String sortStr= bubbleSort(str);
        System.out.println(sortStr);
    }    
}
